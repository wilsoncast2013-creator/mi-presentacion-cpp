        function showExplanation(key) {
            const panel = document.getElementById('explanation-panel');
            panel.innerHTML = explanations[key];
            panel.classList.remove('bg-white');
            panel.classList.add('bg-indigo-50');
            setTimeout(() => {
                panel.classList.remove('bg-indigo-50');
                panel.classList.add('bg-white');
            }, 300);
        }

        function runSimulation() {
            const baseInput = document.getElementById('inputBase').value;
            const alturaInput = document.getElementById('inputAltura').value;
            const consoleDiv = document.getElementById('consoleOutput');
            
            const base = parseFloat(baseInput);
            const altura = parseFloat(alturaInput);
            
            consoleDiv.innerHTML = '';
            
            if (isNaN(base) || isNaN(altura) || base <= 0 || altura <= 0) {
                consoleDiv.innerHTML = '<span class="text-red-500">Error: Por favor ingrese valores numéricos válidos mayores a cero.</span>';
                return;
            }

            const perimetro = 2 * (base + altura);

            const out1 = document.createElement('div');
            out1.innerHTML = `> Ingrese la base: <span class="text-white">${base}</span>`;
            consoleDiv.appendChild(out1);

            setTimeout(() => {
                const out2 = document.createElement('div');
                out2.innerHTML = `> Ingrese la altura: <span class="text-white">${altura}</span>`;
                consoleDiv.appendChild(out2);

                setTimeout(() => {
                    const out3 = document.createElement('div');
                    out3.className = "mt-2 font-bold";
                    out3.innerHTML = `> Resultado: El perimetro es: <span class="text-yellow-300 text-lg">${perimetro}</span>`;
                    consoleDiv.appendChild(out3);
                    
                    const out4 = document.createElement('div');
                    out4.className = "mt-1 text-slate-500 text-xs";
                    out4.innerHTML = `[Proceso completado con código de retorno 0]`;
                    consoleDiv.appendChild(out4);

                    consoleDiv.scrollTop = consoleDiv.scrollHeight;

                    if(simChart) {
                        simChart.data.datasets[0].data = [base, altura, perimetro];
                        simChart.update();
                    }
                }, 400);
            }, 400);
        }

        document.addEventListener('DOMContentLoaded', () => {
            initChart();
        });
